```swift
import Foundation

public enum KeyEncryption: Equatable, Sendable {
    case encrypted
    case unencrypted
    /// Not a recognizable private key head, or no private key at all.
    case unknown
}

/// A `.pub` file's first line: `type base64 [comment]`.
public struct PublicKeyInfo: Equatable, Sendable {
    public let type: String
    public let base64: String
    public let comment: String?
    public let fingerprint: String

    public init?(text: String) {
        let lines = text.unicodeScalars.split(whereSeparator: { $0 == "\n" || $0 == "\r" }).map { String(String.UnicodeScalarView($0)) }
        guard let line = lines.first(where: { !$0.trimmingCharacters(in: .whitespaces).isEmpty }) else { return nil }
        let parts = line.split(whereSeparator: { $0 == " " || $0 == "\t" }).map(String.init)
        guard parts.count >= 2, let fingerprint = Fingerprint.sha256(base64Key: parts[1]) else { return nil }
        type = parts[0]
        base64 = parts[1]
        comment = parts.count > 2 ? parts[2...].joined(separator: " ") : nil
        self.fingerprint = fingerprint
    }
}

/// Looks only at the first bytes of a private key file (header and first base64 line); the key
/// material is never parsed or kept (spec §8.1).
public enum KeyFileInspector {
    public static let headLimit = 512

    public static func readHead(of url: URL) throws -> Data {
        let handle = try FileHandle(forReadingFrom: url)
        defer { try? handle.close() }
        return try handle.read(upToCount: headLimit) ?? Data()
    }

    /// Lines split on LF bytes, a trailing CR removed.
    static func lines(_ head: Data) -> [String] {
        head.split(separator: 0x0A, omittingEmptySubsequences: false).map { slice in
            var bytes = Array(slice)
            if bytes.last == 0x0D { bytes.removeLast() }
            return String(decoding: bytes, as: UTF8.self)
        }
    }

    public static func isPrivateKeyHeader(_ head: Data) -> Bool {
        guard let first = lines(head).first else { return false }
        return first.hasPrefix("-----BEGIN ") && first.hasSuffix("PRIVATE KEY-----")
    }

    public static func encryption(ofPrivateKeyHead head: Data) -> KeyEncryption {
        guard isPrivateKeyHeader(head) else { return .unknown }
        let lines = lines(head)
        switch lines[0] {
        case "-----BEGIN OPENSSH PRIVATE KEY-----":
            guard lines.count > 1, let cipher = openSSHCipher(firstBase64Line: lines[1]) else { return .unknown }
            return cipher == "none" ? .unencrypted : .encrypted
        case "-----BEGIN ENCRYPTED PRIVATE KEY-----":
            return .encrypted
        case "-----BEGIN PRIVATE KEY-----":
            return .unencrypted
        default:   // traditional PEM: RSA / EC / DSA
            guard lines.count > 1 else { return .unknown }
            return lines[1].hasPrefix("Proc-Type:") && lines[1].contains("ENCRYPTED") ? .encrypted : .unencrypted
        }
    }

    /// openssh-key-v1: magic "openssh-key-v1\0", then a uint32 (big endian) length and the cipher name.
    static func openSSHCipher(firstBase64Line line: String) -> String? {
        let usable = String(line.prefix(line.utf8.count / 4 * 4))
        guard let data = Data(base64Encoded: usable) else { return nil }
        let bytes = Array(data)
        let magic = Array("openssh-key-v1".utf8) + [0]
        let lengthStart = magic.count
        let nameStart = lengthStart + 4
        guard bytes.count >= nameStart, Array(bytes[0..<magic.count]) == magic else { return nil }
        let length = bytes[lengthStart..<nameStart].reduce(0) { $0 << 8 | Int($1) }
        guard (1...64).contains(length), bytes.count >= nameStart + length else { return nil }
        return String(decoding: bytes[nameStart..<nameStart + length], as: UTF8.self)
    }
}
```

```swift
        self.publicKey = publicKey
        self.encryption = encryption

/// Finds keys (spec §8.1): a private key is a file whose head is a `-----BEGIN … PRIVATE KEY-----`
/// header and that has a `<name>.pub` beside it; a `.pub` alone is listed as a public-only key.
/// IdentityFile targets outside the scanned directories are added even without a `.pub`.
public struct KeyScanner: Sendable {
    private let runner: any ProcessRunning
    private let sshKeygen: URL

    public init(runner: any ProcessRunning = ProcessRunner(), sshKeygen: URL = URL(filePath: "/usr/bin/ssh-keygen")) {
        self.runner = runner
        self.sshKeygen = sshKeygen
    }

    public func scan(directories: [URL], extraPrivateKeys: [URL] = []) async -> [SSHKey] {
        var keys = Self.discover(directories: directories, extraPrivateKeys: extraPrivateKeys)
        for index in keys.indices {
            guard let pub = keys[index].publicURL,
                  let result = try? await runner.run(sshKeygen, arguments: ["-lf", pub.path], environment: [:], timeout: .seconds(30)),
                  result.exitCode == 0, let described = Self.parseKeygenList(result.stdout) else { continue }
            keys[index].bits = described.bits
            keys[index].typeLabel = described.type
        }
        return keys
    }

    public static func discover(directories: [URL], extraPrivateKeys: [URL]) -> [SSHKey] {
        let fm = FileManager.default
        var seenDirectories = Set<String>()
        var seenFiles = Set<String>()
        var keys: [SSHKey] = []

        func isFile(_ url: URL) -> Bool {
            var isDirectory: ObjCBool = false
            return fm.fileExists(atPath: url.path, isDirectory: &isDirectory) && !isDirectory.boolValue
        }
        func publicInfo(_ url: URL) -> PublicKeyInfo? {
            (try? String(contentsOf: url, encoding: .utf8)).flatMap(PublicKeyInfo.init(text:))
        }
        func addPrivate(_ url: URL) {
            let canonical = KeyUsage.canonicalPath(url)
            guard !seenFiles.contains(canonical), isFile(url),
                  let head = try? KeyFileInspector.readHead(of: url), KeyFileInspector.isPrivateKeyHeader(head) else { return }
            seenFiles.insert(canonical)
            let pub = URL(filePath: url.path + ".pub")
            let info = publicInfo(pub)
            keys.append(SSHKey(privateURL: url, publicURL: info == nil ? nil : pub, publicKey: info,
                               encryption: KeyFileInspector.encryption(ofPrivateKeyHead: head)))
        }

        for directory in directories {
            guard seenDirectories.insert(KeyUsage.canonicalPath(directory)).inserted,
                  let names = try? fm.contentsOfDirectory(atPath: directory.path) else { continue }
            let visible = names.filter { !$0.hasPrefix(".") }.sorted()
            for name in visible where !name.hasSuffix(".pub") && visible.contains(name + ".pub") {
                addPrivate(directory.appending(path: name))
            }
            for name in visible where name.hasSuffix(".pub") && !name.hasSuffix("-cert.pub") {
                let pub = directory.appending(path: name)
                let base = URL(filePath: String(pub.path.dropLast(4)))
                guard !seenFiles.contains(KeyUsage.canonicalPath(base)), isFile(pub), let info = publicInfo(pub) else { continue }
                seenFiles.insert(KeyUsage.canonicalPath(pub))
                keys.append(SSHKey(privateURL: nil, publicURL: pub, publicKey: info, encryption: .unknown))
            }
        }
        for url in extraPrivateKeys { addPrivate(url) }
        return keys.sorted { $0.name.utf8.lexicographicallyPrecedes($1.name.utf8) }
    }

    /// `256 SHA256:… comment (ED25519)` → (256, "ED25519").
    public static func parseKeygenList(_ output: String) -> (bits: Int, type: String)? {
        guard let line = output.split(whereSeparator: \.isNewline).first else { return nil }
        let parts = line.split(separator: " ")
        guard let first = parts.first, let bits = Int(first), let last = parts.last,
              last.hasPrefix("("), last.hasSuffix(")"), last.count > 2 else { return nil }
        return (bits, String(last.dropFirst().dropLast()))
    }
}
```

```swift
import Foundation

/// Which hosts use a key: IdentityFile values resolved the way they name files at connect time.
public enum KeyUsage {
    /// `~/…` and `%d/…` → home; a relative path → home-relative (ssh is started from the home
    /// directory). nil for other `%` tokens or `~user` forms, which cannot be resolved here.
    public static func resolve(_ value: String, home: URL) -> URL? {
        var path = value
        if path == "~" || path.hasPrefix("~/") {
            path = home.path + path.dropFirst(1)
        } else if path.hasPrefix("%d/") {
            path = home.path + path.dropFirst(2)
        }
        guard !path.contains("%"), !path.hasPrefix("~") else { return nil }
        let url = path.hasPrefix("/") ? URL(filePath: path) : home.appending(path: path)
        return url.standardizedFileURL
    }

    public static func canonicalPath(_ url: URL) -> String {
        url.resolvingSymlinksInPath().standardizedFileURL.path
    }

    /// Every resolvable IdentityFile of the given hosts, once.
    public static func identityFiles(of hosts: [HostEntry], home: URL) -> [URL] {
        var seen = Set<String>()
        var result: [URL] = []
        for host in hosts {
            for value in host.values(for: "IdentityFile") {
                guard let url = resolve(value, home: home), seen.insert(canonicalPath(url)).inserted else { continue }
                result.append(url)
            }
        }
        return result
    }

    /// Host labels (patterns joined by a space, file order) whose IdentityFile names `key`.
    public static func hosts(using key: URL, in hosts: [HostEntry], home: URL) -> [String] {
        let target = canonicalPath(key)
        var result: [String] = []
        for host in hosts {
            let uses = host.values(for: "IdentityFile").contains { value in
                resolve(value, home: home).map { canonicalPath($0) == target } ?? false
            }
            let label = host.patterns.joined(separator: " ")
            if uses && !result.contains(label) { result.append(label) }
        }
        return result
    }
}
```

```swift
import Foundation

/// A one-shot SSH_ASKPASS helper (spec §8.3, §12.3), written per operation into a private 0700
/// directory. The script holds no secret: the passphrase is passed only in the child's environment.
/// It answers once; a second prompt (wrong passphrase) fails, so ssh-add stops asking.
struct AskPass {
    let directory: URL
    var helper: URL { directory.appending(path: "askpass") }
    var stateFile: URL { directory.appending(path: "answered") }

    static let script = """
    #!/bin/sh
    if [ -e "$SSHCM_ASKPASS_STATE" ]; then exit 1; fi
    : > "$SSHCM_ASKPASS_STATE"
    printf '%s\\n' "$SSHCM_ASKPASS_SECRET"

    """

    static func make(in parent: URL) throws -> AskPass {
        let fm = FileManager.default
        let directory = parent.appending(path: "sshcm-askpass-\(UUID().uuidString)", directoryHint: .isDirectory)
        try fm.createDirectory(at: directory, withIntermediateDirectories: false, attributes: [.posixPermissions: 0o700])
        let askPass = AskPass(directory: directory)
        guard fm.createFile(atPath: askPass.helper.path, contents: Data(script.utf8), attributes: [.posixPermissions: 0o700]) else {
            askPass.remove()
            throw CocoaError(.fileWriteUnknown, userInfo: [NSFilePathErrorKey: askPass.helper.path])
        }
        try fm.setAttributes([.posixPermissions: 0o700], ofItemAtPath: askPass.helper.path)   // umask may have narrowed it
        return askPass
    }

    func environment(secret: String) -> [String: String] {
        [
            "SSH_ASKPASS": helper.path,
            "SSH_ASKPASS_REQUIRE": "force",
            "SSHCM_ASKPASS_STATE": stateFile.path,
            "SSHCM_ASKPASS_SECRET": secret,
        ]
    }

    var wasAsked: Bool { FileManager.default.fileExists(atPath: stateFile.path) }

    func remove() {
        try? FileManager.default.removeItem(at: directory)
    }
}
```

```swift
import Foundation

public enum SSHKeyType: String, CaseIterable, Sendable {
    case ed25519
    case rsa   // always 4096 bits

    var arguments: [String] {
        switch self {
        case .ed25519: ["-t", "ed25519"]
        case .rsa: ["-t", "rsa", "-b", "4096"]
        }
    }

    public var defaultFileName: String {
        switch self {
        case .ed25519: "id_ed25519"
        case .rsa: "id_rsa"
        }
    }
}

public enum KeyManagerError: Error, Equatable, Sendable {
    case fileExists(String)
    /// ssh-keygen refuses passphrases shorter than five bytes.
    case passphraseTooShort
    case wrongPassphrase
    case noAgent
    case timedOut
    /// ssh's first stderr line (English, untranslated by design).
    case failed(String)
}

/// ssh-keygen / ssh-add operations (spec §8.3). Passphrases appear only in ssh-keygen's `-N`/`-P`
/// arguments and the askpass child's environment (accepted trade-off, spec §8.4); never logged or stored.
public struct KeyManager: Sendable {
    public static let minimumPassphraseLength = 5

    private let runner: any ProcessRunning
    private let toolsDirectory: URL
    private let agentSocket: String?
    private let askPassDirectory: URL

    /// `agentSocket` nil means the user's agent (inherited `SSH_AUTH_SOCK`); tests pass their own.
    public init(runner: any ProcessRunning = ProcessRunner(),
                toolsDirectory: URL = URL(filePath: "/usr/bin"),
                agentSocket: String? = nil,
                askPassDirectory: URL = FileManager.default.temporaryDirectory) {
        self.runner = runner
        self.toolsDirectory = toolsDirectory
        self.agentSocket = agentSocket
        self.askPassDirectory = askPassDirectory
    }

    public static func isValidPassphrase(_ passphrase: String) -> Bool {
        passphrase.isEmpty || passphrase.utf8.count >= minimumPassphraseLength
    }

    /// Never overwrites: fails if the key or its `.pub` exists. Creates the directory (0700) if needed.
    public func generate(type: SSHKeyType, at url: URL, comment: String, passphrase: String) async throws {
        let fm = FileManager.default
        for path in [url.path, url.path + ".pub"] where fm.fileExists(atPath: path) {
            throw KeyManagerError.fileExists(path)
        }
        guard Self.isValidPassphrase(passphrase) else { throw KeyManagerError.passphraseTooShort }
        let directory = url.deletingLastPathComponent()
        if !fm.fileExists(atPath: directory.path) {
            try fm.createDirectory(at: directory, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
        }
        let result = try await run("ssh-keygen", ["-q"] + type.arguments + ["-C", comment, "-f", url.path, "-N", passphrase])
        try check(result)
    }

    public func changePassphrase(at url: URL, old: String, new: String) async throws {
        guard Self.isValidPassphrase(new) else { throw KeyManagerError.passphraseTooShort }
        let result = try await run("ssh-keygen", ["-p", "-f", url.path, "-P", old, "-N", new])
        if result.exitCode != 0 && result.stderr.contains("incorrect passphrase") { throw KeyManagerError.wrongPassphrase }
        try check(result)
    }

    public func addToAgent(_ url: URL, passphrase: String?, useKeychain: Bool) async throws {
        let askPass = try AskPass.make(in: askPassDirectory)
        defer { askPass.remove() }
        let arguments = (useKeychain ? ["--apple-use-keychain"] : []) + [url.path]
        let environment = agentEnvironment.merging(askPass.environment(secret: passphrase ?? "")) { _, new in new }
        let result = try await run("ssh-add", arguments, environment: environment)
        guard result.exitCode != 0 else { return }
        if result.exitCode == 2 || result.stderr.contains("Error connecting to agent")
            || result.stderr.contains("Could not open a connection") {
            throw KeyManagerError.noAgent
        }
        if askPass.wasAsked { throw KeyManagerError.wrongPassphrase }
        throw KeyManagerError.failed(ConnectionClassifier.firstLine(result.stderr))
    }

    /// Fingerprints loaded in the agent; empty if it has none; nil if no agent is reachable.
    public func agentFingerprints() async -> Set<String>? {
        guard let result = try? await run("ssh-add", ["-l"], environment: agentEnvironment) else { return nil }
        switch result.exitCode {
        case 0: return Self.parseAgentList(result.stdout)
        case 1: return []
        default: return nil
        }
    }

    public static func parseAgentList(_ output: String) -> Set<String> {
        Set(output.split(whereSeparator: \.isNewline).compactMap { line in
            line.split(separator: " ").first { $0.hasPrefix("SHA256:") }.map(String.init)
        })
    }

    /// `fileName`, else `fileName_2`, `_3` … — a name for which neither the key nor its `.pub` exists.
    public static func nextAvailableURL(fileName: String, in directory: URL) -> URL {
        let fm = FileManager.default
        var candidate = directory.appending(path: fileName)
        var counter = 2
        while fm.fileExists(atPath: candidate.path) || fm.fileExists(atPath: candidate.path + ".pub") {
            candidate = directory.appending(path: "\(fileName)_\(counter)")
            counter += 1
        }
        return candidate
    }

    private var agentEnvironment: [String: String] {
        agentSocket.map { ["SSH_AUTH_SOCK": $0] } ?? [:]
    }

    private func run(_ tool: String, _ arguments: [String], environment: [String: String] = [:]) async throws -> ProcessResult {
        let result = try await runner.run(toolsDirectory.appending(path: tool), arguments: arguments,
                                          environment: environment, timeout: .seconds(30))
        if result.timedOut { throw KeyManagerError.timedOut }
        return result
    }

    private func check(_ result: ProcessResult) throws {
        guard result.exitCode == 0 else { throw KeyManagerError.failed(ConnectionClassifier.firstLine(result.stderr)) }
    }
}
```

```swift
import Foundation
import Observation
import SSHConfigKit

struct KeyRow: Identifiable, Equatable {
    let key: SSHKey
    /// Hosts of the saved file whose IdentityFile names this key.
    let usedBy: [String]
    /// nil: the agent could not be asked.
    let inAgent: Bool?
    let displayPath: String

    var id: String { key.id }
}

@MainActor
@Observable
final class KeysModel {
    private(set) var rows: [KeyRow] = []
    private(set) var isRefreshing = false
    private(set) var isWorking = false
    var errorMessage: String?
    var selection: Set<String> = []

    @ObservationIgnored let tools: KeyTools
    @ObservationIgnored weak var app: AppModel?
    @ObservationIgnored private var refreshGeneration = 0

    init(tools: KeyTools) {
        self.tools = tools
    }

    var defaultComment: String {
        "\(NSUserName())@\(ProcessInfo.processInfo.hostName)"
    }

    func suggestedFileName(for type: SSHKeyType) -> String {
        guard let app else { return type.defaultFileName }
        return KeyManager.nextAvailableURL(fileName: type.defaultFileName, in: app.sshDirectory).lastPathComponent
    }

    func refresh() async {
        guard let app else { return }
        refreshGeneration += 1
        let generation = refreshGeneration
        isRefreshing = true
        defer { if generation == refreshGeneration { isRefreshing = false } }
        let hosts = app.savedHostEntries
        let home = app.settings.homeDirectory
        let keys = await tools.scanner.scan(directories: app.keyDirectories,
                                            extraPrivateKeys: KeyUsage.identityFiles(of: hosts, home: home))
        let agent = await tools.manager.agentFingerprints()
        guard generation == refreshGeneration else { return }
        rows = keys.map { key in
            KeyRow(
                key: key,
                usedBy: KeyUsage.hosts(using: key.fileURL, in: hosts, home: home),
                inAgent: agent.map { loaded in key.publicKey.map { loaded.contains($0.fingerprint) } ?? false },
                displayPath: app.settings.displayPath(for: key.fileURL))
        }
        selection = selection.filter { id in rows.contains { $0.id == id } }
    }

    /// Creates `~/.ssh/<fileName>`; optionally adds it to a host as an unsaved edit (saved by the Save flow).
    func generate(type: SSHKeyType, fileName: String, comment: String, passphrase: String, assignTo alias: String?) async -> Bool {
        guard let app else { return false }
        guard Self.isValidFileName(fileName) else {
            errorMessage = String(localized: "keys.error.invalidFileName")
            return false
        }
        let url = app.sshDirectory.appending(path: fileName)
        return await perform {
            try await tools.manager.generate(type: type, at: url, comment: comment, passphrase: passphrase)
            if let alias { try app.assignIdentityFile(url, toAlias: alias) }
        }
    }

    func addToAgent(_ row: KeyRow, passphrase: String?) async -> Bool {
        guard let url = row.key.privateURL else { return false }
        return await perform {
            try await tools.manager.addToAgent(url, passphrase: passphrase, useKeychain: tools.useKeychain)
        }
    }

    func changePassphrase(_ row: KeyRow, old: String, new: String) async -> Bool {
        guard let url = row.key.privateURL else { return false }
        return await perform { try await tools.manager.changePassphrase(at: url, old: old, new: new) }
    }

    func copyPublicKey(_ row: KeyRow) {
        guard let url = row.key.publicURL, let text = try? String(contentsOf: url, encoding: .utf8) else { return }
        tools.copyToPasteboard(text.trimmingCharacters(in: .whitespacesAndNewlines))
    }

    /// Moves the private key and its `.pub` to the Trash together (spec §8.3).
    func trash(_ row: KeyRow) async -> Bool {
        let urls = [row.key.privateURL, row.key.publicURL].compactMap { $0 }
        return await perform { try await tools.recycle(urls) }
    }

    /// Runs an operation, reports errors, then refreshes the list.
    private func perform(_ operation: () async throws -> Void) async -> Bool {
        isWorking = true
        defer { isWorking = false }
        do {
            try await operation()
            errorMessage = nil
            await refresh()
            return true
        } catch {
            errorMessage = Messages.text(for: error)
            return false
        }
    }

    static func isValidFileName(_ name: String) -> Bool {
        !name.isEmpty && !name.hasPrefix(".") && !name.contains("/")
            && !name.unicodeScalars.contains { $0.value < 0x20 || $0.value == 0x7F }
    }
}
```

```swift
            comment = keys.defaultComment
        }
```

```swift
import SSHConfigKit
import SwiftUI

struct KeysWindow: View {
    let app: AppModel
    @State private var sheet: KeySheet?
    @State private var trashing: KeyRow?

    enum KeySheet: Identifiable {
        case new, addToAgent(KeyRow), changePassphrase(KeyRow), upload(KeyRow)
        var id: String {
            switch self {
            case .new: "new"
            case .addToAgent(let row): "agent-" + row.id
            case .changePassphrase(let row): "pass-" + row.id
            case .upload(let row): "upload-" + row.id
            }
        }
    }

    private var keys: KeysModel { app.keys }

    var body: some View {
        @Bindable var keys = app.keys
        Table(keys.rows, selection: $keys.selection) {
            TableColumn(String(localized: "keys.column.name")) { row in
                HStack(spacing: 4) {
                    if !row.key.hasPrivateKey {
                        Image(systemName: "key.slash").foregroundStyle(.secondary)
                            .help(String(localized: "keys.noPrivateKey"))
                            .accessibilityLabel(String(localized: "keys.noPrivateKey"))
                    }
                    Text(verbatim: row.displayPath).lineLimit(1).help(row.displayPath)
                }
            }
            TableColumn(String(localized: "keys.column.type")) { row in
                let bits = row.key.bits.map(String.init) ?? ""
                Text(verbatim: [row.key.typeLabel ?? row.key.publicKey?.type ?? "", bits].filter { !$0.isEmpty }.joined(separator: " "))
            }
            .width(min: 70, ideal: 100)
            TableColumn(String(localized: "keys.column.fingerprint")) { row in
                Text(verbatim: row.key.publicKey?.fingerprint ?? "").font(.caption.monospaced()).lineLimit(1).textSelection(.enabled)
            }
            .width(min: 180, ideal: 320)
            TableColumn(String(localized: "keys.column.comment")) { row in Text(verbatim: row.key.publicKey?.comment ?? "") }
            TableColumn(String(localized: "keys.column.passphrase")) { row in passphraseCell(row) }
                .width(min: 60, ideal: 80)
            TableColumn(String(localized: "keys.column.agent")) { row in agentCell(row) }
                .width(min: 60, ideal: 80)
            TableColumn(String(localized: "keys.column.usedBy")) { row in
                Text(verbatim: row.usedBy.joined(separator: ", ")).lineLimit(1).help(row.usedBy.joined(separator: "\n"))
            }
        }
        .contextMenu(forSelectionType: String.self) { ids in
            if let row = keys.rows.first(where: { ids.contains($0.id) }) { actions(for: row) }
        }
        .overlay {
            if keys.rows.isEmpty && !keys.isRefreshing {
                ContentUnavailableView(String(localized: "keys.empty"), systemImage: "key")
            }
        }
        .toolbar {
            ToolbarItem(placement: .primaryAction) {
                Button { sheet = .new } label: { Label(String(localized: "keys.new"), systemImage: "plus") }
            }
            ToolbarItem(placement: .primaryAction) {
                Button { Task { await keys.refresh() } } label: {
                    Label(String(localized: "keys.refresh"), systemImage: "arrow.clockwise")
                }
                .disabled(keys.isRefreshing)
            }
            ToolbarItem(placement: .primaryAction) {
                Menu {
                    if let row = selectedRow { actions(for: row) }
                } label: {
                    Label(String(localized: "keys.title"), systemImage: "ellipsis.circle")
                }
                .disabled(selectedRow == nil)
            }
        }
        .sheet(item: $sheet) { item in
            switch item {
            case .new: NewKeySheet(keys: keys, assignableAliases: app.assignableAliases) { sheet = nil }
            case .addToAgent(let row): AddToAgentSheet(keys: keys, row: row) { sheet = nil }
            case .changePassphrase(let row): ChangePassphraseSheet(keys: keys, row: row) { sheet = nil }
            case .upload(let row): UploadKeySheet(app: app, row: row) { sheet = nil }
            }
        }
        .confirmationDialog(
            String(localized: "keys.trash.title"),
            isPresented: Binding(get: { trashing != nil }, set: { if !$0 { trashing = nil } }),
            presenting: trashing
        ) { row in
            Button(String(localized: "keys.trash"), role: .destructive) { Task { _ = await keys.trash(row) } }
        } message: { row in
            let hosts = row.usedBy.joined(separator: ", ")
            Text(row.usedBy.isEmpty ? String(localized: "keys.trash.message")
                 : String(localized: "keys.trash.message") + "\n\n" + String(localized: "keys.trash.inUse \(hosts)"))
        }
        .alert(
            String(localized: "error.title"),
            isPresented: Binding(get: { sheet == nil && keys.errorMessage != nil }, set: { if !$0 { keys.errorMessage = nil } })
        ) {
            Button(String(localized: "action.close")) { keys.errorMessage = nil }
        } message: {
            Text(verbatim: keys.errorMessage ?? "")
        }
        .task { await keys.refresh() }
        .frame(minWidth: 900, minHeight: 420)
    }

    private var selectedRow: KeyRow? {
        keys.rows.first { keys.selection.contains($0.id) }
    }

    @ViewBuilder
    private func actions(for row: KeyRow) -> some View {
        Button(String(localized: "keys.copyPublic")) { keys.copyPublicKey(row) }
            .disabled(row.key.publicURL == nil)
        Button(String(localized: "keys.addToAgent")) {
            if row.key.encryption == .unencrypted {
                Task { _ = await keys.addToAgent(row, passphrase: nil) }
            } else {
                sheet = .addToAgent(row)
            }
        }
        .disabled(!row.key.hasPrivateKey)
        Button(String(localized: "keys.upload")) { sheet = .upload(row) }
            .disabled(row.key.publicURL == nil)
        Button(String(localized: "keys.changePassphrase")) { sheet = .changePassphrase(row) }
            .disabled(!row.key.hasPrivateKey)
        Divider()
        Button(String(localized: "keys.trash"), role: .destructive) { trashing = row }
    }

    @ViewBuilder
    private func passphraseCell(_ row: KeyRow) -> some View {
        switch row.key.encryption {
        case .encrypted:
            Text(String(localized: "keys.passphrase.yes"))
        case .unencrypted:
            Label(String(localized: "keys.passphrase.no"), systemImage: "exclamationmark.triangle.fill")
                .foregroundStyle(.orange)
                .help(String(localized: "keys.passphrase.noHelp"))
        case .unknown:
            Text(verbatim: "—").foregroundStyle(.secondary)
        }
    }

    @ViewBuilder
    private func agentCell(_ row: KeyRow) -> some View {
        switch row.inAgent {
        case .some(true):
            Label(String(localized: "keys.agent.loaded"), systemImage: "checkmark.circle.fill").foregroundStyle(.green)
        case .some(false):
            Text(verbatim: "—").foregroundStyle(.secondary)
        case .none:
            Image(systemName: "questionmark.circle").foregroundStyle(.secondary)
                .help(String(localized: "keys.agent.unknown"))
                .accessibilityLabel(String(localized: "keys.agent.unknown"))
        }
    }
}
```
